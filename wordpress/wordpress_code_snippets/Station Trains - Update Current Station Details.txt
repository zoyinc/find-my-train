<script>
//
// Update the details for the currently selected location
//

// We need to wait until the entire page is loaded so use "window.addEventListener('load'..."
window.addEventListener('load', function () {
	// Get a reference to the title heading
	currTrainTitleElement = document.getElementsByClassName("zoyinc_post_title")[0];
	
	// Get a reference to the current train group
	currTrainGroupElement = document.getElementsByClassName("zoyinc_curr_station_details")[0];
<?php	
	global $wpdb; // Use the existing WordPress DB connection
	$locationName = "Location not determined - Error";
	
	// Get the default location from the config table 
	$query_results = $wpdb->get_results("
		SELECT 
			default_location
		FROM 
			fmt_config fc
		;");
	// Need to set defaults in case the fmt_config table is empty
	$config_default_location = "1";
	foreach($query_results as $curr_result){
		$config_default_location = $curr_result->default_location;
	};
		
	//
	// Get current location number
	// The url parameter 'location_number' can set the location - this is done from the dropdown on the page
	// However if this is the first time the page is visited then it will be unset, at
	// which point it should use the '$config_default_location' variable, which is set in the config above
	// and extracted from the DB earlier
	// 
	$curr_location_number = get_query_var('location_number');
	// if $curr_location_number is not set, use the default
	if ($curr_location_number == "") {
		$curr_location_number = $config_default_location;
	};
	
	// Get the location name 
	// The "source of truth" for this is the table fmt_track_sections
	$query_results = $wpdb->get_results("
	SELECT 
		* 
	FROM 
		fmt_track_sections 
	WHERE 
		id = " . $curr_location_number . "
	;");
	foreach($query_results as $curr_result){
		$locationName = $curr_result->title;
		$primaryImageURL = $curr_result->primary_image;
		$secondaryImageURL = $curr_result->secondary_image ;
		$locationDescription = $curr_result->description ;
		$locationIsFeatured = $curr_result->featured ;
	};
		
	//
	// The location information ultimately comes from the file "stations.csv".
	// 
	// This file only has details about "featured" locations, so there is a huge
	// number of locations which do not have special information and need generic
	// details. To accomodate this we specify a "generic location" in stations.csv with an id of "-1"
	// 
	
	// Get the details for the generic location
	$query_results = $wpdb->get_results("
	SELECT 
		* 
	FROM 
		fmt_stations 
	WHERE 
		section_id = -1
	;");
	foreach($query_results as $curr_result){
		$primaryImageURL = $curr_result->primary_image;
		$secondaryImageURL = $curr_result->secondary_image ;
		$locationDescription = $curr_result->description ;
		$locationIsFeatured = $curr_result->featured ;
	};
	
	// Get the details for the current location
	$query_results = $wpdb->get_results("
	SELECT 
		* 
	FROM 
		fmt_stations 
	WHERE 
		section_id = " . $curr_location_number . "
	;");
	foreach($query_results as $curr_result){
		$primaryImageURL = $curr_result->primary_image;
		$secondaryImageURL = $curr_result->secondary_image ;
		$locationDescription = $curr_result->description ;
		$locationIsFeatured = $curr_result->featured ;
	};
	// Update the post title
	echo "currTrainTitleElement.innerHTML = \"Trains at " . $locationName . "\"; \n";
	
	//
	// Update the featured/primary image
	// 
	echo "featuredImageFigureElement = document.getElementsByClassName(\"zoyinc_featured_image_single\")[0]; \n"; 
	echo "featuredImgElement = featuredImageFigureElement.getElementsByTagName(\"img\")[0]; \n";
	echo "featuredImgElement.src= \"" . $primaryImageURL . "\"; \n";
	echo "featuredImgElement.srcset= \"" . $primaryImageURL . "\" + \" 1200w\"; \n";
	echo "delete featuredImgElement.style.removeProperty('aspect-ratio'); \n";
	
	//
	// Update the description
	// 
	// The description can potentially look like:
	// 
	//    "lots of rubbish and junk<image>with lots of useful stuff"
	//    
	// In other words there could be an "<image>" tag in the description, that means that the locations
	// seconday image should be displayed at that point.
	// 
	// So what we do is look for an <image> tag and if it exists we split the description into
	// 2 variables locationDescriptionTop and locationDescriptionBottom, where the "top" description is
	// displayed before the image, and the "bottom" is displayed after the image.
	// Obviously if there is no <image> tag the full description goes into top.
	// 
	if (strpos($locationDescription, '<image>') !== false) {
		$hasImage = 'Y';

		// Split the string at <image>
		$parts = explode('<image>', $locationDescription, 2);
		$locationDescriptionTop = $parts[0];
		$locationDescriptionBottom = isset($parts[1]) ? $parts[1] : '';
	} else {
		$hasImage = 'N';
		$locationDescriptionTop = $locationDescription;
		$locationDescriptionBottom = '';
	}
	// Get a reference to the current location group
	echo "currLocationGroupElement = document.getElementsByClassName(\"zoyinc_curr_station_details\")[0]; \n";
	
	// Get references to the current description blocks
	echo "currLocationDescriptionTop = currLocationGroupElement.getElementsByClassName(\"zoyinc_curr_train_description_top\")[0]; \n";
	echo "currLocationDescriptionBottom = currLocationGroupElement.getElementsByClassName(\"zoyinc_curr_train_description_bottom\")[0]; \n";
	
	// Update the description
	echo "currLocationDescriptionTop.innerHTML = \"" . $locationDescriptionTop . "\"; \n";
	echo "currLocationDescriptionBottom.innerHTML = \"" . $locationDescriptionBottom . "\"; \n";
	
	//
	// Update the secondary image
	// 		
	if ($hasImage == 'Y') {
		echo "secondaryImgElement = currLocationGroupElement.getElementsByTagName(\"img\")[0]; \n";
		echo "secondaryImgElement.src= \"" . $secondaryImageURL . "\"; \n";
		echo "secondaryImgElement.srcset= \"" . $secondaryImageURL . "\" + \" 1200w\"; \n";
		echo "delete secondaryImgElement.style.removeProperty('aspect-ratio'); \n";
	} else {
		echo "currSecondImageRow = currLocationGroupElement.getElementsByClassName(\"zoyinc_curr_location_second_image_row\")[0]; \n";
		echo "currSecondImageRow.remove(); \n";
	}

?>
	
	//
	// Update the trip details table for the current train
	//  
	figureElement = document.getElementsByClassName("zoyinc-find_my_train-cur_train_trips")[0]; 
	curTrainTripTable = figureElement.getElementsByTagName("table")[0];

<?php	
	// Get trip details for the current train
	$query_results = $wpdb->get_results("
		SELECT 
			stop_details_str, 
			trip_delay
		FROM 
			fmt_train_details ftd, 
			fmt_trips ft  
		WHERE 
			train_number = \"" . $curr_train_number . "\" 
			AND ftd.whole_train_trip_id = ft.trip_id
		;");
	
	// 
	// Trip details are contained in the column "stop_details_str"
	// This is a semicolon separated list of platform details. This looks something like:
	// 
	//    "1,Britomart Train Station 2,20280,05:38:00;2,Orakei Train Station 2,20640,05:44:00; ..."
	//    
	// Each set of platform details thus looks like:
	// 
	//    "1,Britomart Train Station 2,20280,05:38:00;"
	//    
	// Thus the platform details is a colon separated list
	// 
	
	// First get the "stop_details_str" and "train_featured_img_url" 
	$trip_details = "";
 	foreach($query_results as $curr_result){
		$trip_details = $curr_result->stop_details_str ;
		
	};
	
	
	
	// Check there are trip details for this train
	if ($trip_details != ""){
		$trip_rows = explode(";", $trip_details);
		$currRowNumber = 1;
		
		// Iterate over each set of platform details
		foreach ($trip_rows as $currRow) {
			
			// Split the platfrom details into an array
			$tripRowColumns = explode(",", $currRow);
			
			// The table needs to come with one row of data, otherwise the formatting
			// gets messed up - blame WordPress
			// We add a row for each set of platform details, but since the table
			// comes with the first row already present we only insert a row if it's not the
			// first set of platorm details
			if ( $currRowNumber > 1 ){
				echo "newRow = curTrainTripTable.insertRow(curTrainTripTable.rows.length);";
				echo "newRow.insertCell(0);";
				echo "newRow.insertCell(1);";
			};
			
			// We are going to preset the depart time as being the scheduled time plus how
			// much the train is delay, or early. So the user only sees the ETA.
			// The most practical way of doing this is in the front end as the delay can go up
			// and down quite a bit during a trip, but the scheduled times don't change.
			$tripDelay = $curr_result->trip_delay;
			$estimatedDepartTime = $tripRowColumns[2] + $tripDelay;
			
			// It was challenging to work out how to format the time string, I looked at various JavaScript
			// functions, but they all had issues.
			// Then I realised it was much simpler to do it "manually" as my requirements were quite simple.
			$estimatedHour = intval($estimatedDepartTime/3600);
			$estimatedMin = round((($estimatedDepartTime - ($estimatedHour*3600))/60),0);
			if ($estimatedHour > 24){
				$estimatedHour = $estimatedHour -24;
			};
			$tripStrSuffix = "am";
			if ($estimatedHour > 12){
				$estimatedHour = $estimatedHour - 12;
				$tripStrSuffix = "pm";
			}
			$estimatedMinStr = (string)$estimatedMin;
			if ($estimatedMin < 10){
				$estimatedMinStr = "0" . (string)$estimatedMin;
			}
			
			echo "curTrainTripTable.rows[" . $currRowNumber . "].cells[0].innerHTML = \"" . $tripRowColumns[1] . "\";";
			echo "curTrainTripTable.rows[" . $currRowNumber . "].cells[1].innerHTML = \"" . $estimatedHour . ":" . $estimatedMinStr . " " . $tripStrSuffix . "\";";
			$currRowNumber+= 1;
		};
	};
?>
	//
	// Update the Featured image
	// 
	featuredImageFigureElement = document.getElementsByClassName("zoyinc_featured_image_single")[0]; 
	featuredImgElement = featuredImageFigureElement.getElementsByTagName("img")[0];
	featuredImgElement.src= trainFeaturedImgURLRelative;
	featuredImgElement.srcset= trainFeaturedImgURLRelative + " 1200w";
	delete featuredImgElement.style.removeProperty('aspect-ratio');
	


});	

</script>
